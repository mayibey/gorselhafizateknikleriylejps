  const W = 1080, H = 1920;
  const cv = document.createElement('canvas'); cv.width = W; cv.height = H;
  const c = cv.getContext('2d');
  const gr = c.createLinearGradient(0, 0, 0, H);
  gr.addColorStop(0, '#0c2740'); gr.addColorStop(.55, '#08203a'); gr.addColorStop(1, '#050f1e');
  c.fillStyle = gr; c.fillRect(0, 0, W, H);
  const rg = c.createRadialGradient(W / 2, -120, 60, W / 2, -120, 900);
  rg.addColorStop(0, 'rgba(240,183,51,.20)'); rg.addColorStop(1, 'rgba(240,183,51,0)');
  c.fillStyle = rg; c.fillRect(0, 0, W, 760);
  c.fillStyle = 'rgba(255,255,255,.10)';
  for (let i = 0; i < 90; i++) { const x = (i * 137.5) % W, yv = (i * 263.9) % 760; c.beginPath(); c.arc(x, yv, i % 7 === 0 ? 2.2 : 1.1, 0, 7); c.fill(); }

  c.fillStyle = '#F3C24A'; c.font = '700 42px Georgia,serif';
  c.fillText('MEVZU JSPS', W / 2, 190);
  c.strokeStyle = 'rgba(240,183,51,.55)'; c.lineWidth = 3;

  c.font = '124px system-ui'; c.textBaseline = 'middle'; c.fillStyle = '#fff';
  c.fillText(PAYLAS_IKON[o.id] || o.ikon || '🎮', W / 2, 488); c.textBaseline = 'alphabetic';
  c.fillStyle = '#F2F7FA'; c.font = '700 84px Georgia,serif';
  wrap(c, o.ad || 'Oyun', W / 2, 730, 920, 90);
  let y0 = 860;
  if (o.yildiz != null) {
    c.font = '96px system-ui';
    const bw = 96; const bas = W / 2 - (3 * bw) / 2 + bw / 2;
    for (let i = 0; i < 3; i++) { c.fillStyle = i < o.yildiz ? '#F3C24A' : 'rgba(255,255,255,.22)'; c.fillText(i < o.yildiz ? '★' : '☆', bas + i * bw, y0 + 30); }

  }
  c.fillStyle = '#F3C24A'; c.font = '800 150px Georgia,serif';

  function wrap(cc, txt, cx, cy, maxw, lh) { const ws = ('' + txt).split(' '); let line = ''; const lines = []; for (const w of ws) { const test = line ? line + ' ' + w : w; if (cc.measureText(test).width > maxw && line) { lines.push(line); line = w; } else line = test; } lines.push(line); const startY = cy - ((lines.length - 1) * lh) / 2; lines.forEach((l, i) => cc.fillText(l, cx, startY + i * lh)); }
